local function addConnection(connection)
    if connection then
        table.insert(connections, connection)
    end
    return connection
end

local function disconnectList(list)
    if not list then
        return
    end
    for _, connection in ipairs(list) do
        pcall(function()
            connection:Disconnect()
        end)
    end
end

local function disconnectAll()
    disconnectList(connections)
    connections = {}

    for player, list in pairs(playerConnections) do
        disconnectList(list)
        playerConnections[player] = nil
    end

    for player, object in pairs(espObjects) do
        if object and object.Destroy then
            pcall(function()
                object:Destroy()
            end)
        end
        espObjects[player] = nil
    end

    pcall(function()
        RunService:UnbindFromRenderStep("AutoGameMenuCameraLock")
    end)
end

local function addCorner(instance, radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius or 8)
    corner.Parent = instance
    return corner
end

local function addStroke(instance, color, thickness, transparency)
    local stroke = Instance.new("UIStroke")
    stroke.Color = color or COLORS.stroke
    stroke.Thickness = thickness or 1
    stroke.Transparency = transparency or 0
    stroke.Parent = instance
    return stroke
end

local function createLabel(parent, text, size, position, textSize, color)
    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.Size = size
    label.Position = position
    label.Text = text
    label.TextColor3 = color or COLORS.text
    label.TextSize = textSize or 13
    label.Font = Enum.Font.Gotham
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = parent
    return label
end

local function createButton(parent, name, size, position, text, color, textSize)
    local button = Instance.new("TextButton")
    button.Name = name
    button.Size = size
    button.Position = position
    button.BackgroundColor3 = color or COLORS.panel2
    button.BorderSizePixel = 0
    button.Text = text
    button.TextColor3 = COLORS.text
    button.TextSize = textSize or 12
    button.Font = Enum.Font.GothamSemibold
    button.AutoButtonColor = false
    button.Parent = parent
    addCorner(button, 7)
    return button
end

-- =========================================================
-- Small hover polish for buttons
-- =========================================================

local function addButtonHover(button, baseTransparency)
    baseTransparency = baseTransparency or 0

    addConnection(button.MouseEnter:Connect(function()
        if button.Parent then
            button.BackgroundTransparency = math.max(0, baseTransparency - 0.06)
        end
    end))

    addConnection(button.MouseLeave:Connect(function()
        if button.Parent then
            button.BackgroundTransparency = baseTransparency
        end
    end))
end

for _, button in ipairs({
    lockButton, teamButton, targetButton, holdKeyButton,
    espButton, auraButton, boxButton, skeletonButton, nameButton,
    hideKeyButton, minimizeButton
}) do
    addButtonHover(button, button.BackgroundTransparency)
end

-- =========================================================
-- 5.1 Initial centered placement / responsive clamp
-- =========================================================

local function clampPanelToViewport()
    local camera = workspace.CurrentCamera
    if not camera then
        return
    end

    local viewport = camera.ViewportSize
    local size = panel.AbsoluteSize
    local halfW = size.X * 0.5
    local halfH = size.Y * 0.5

    local minX = halfW + 8
    local maxX = math.max(minX, viewport.X - halfW - 8)
    local minY = halfH + 8
    local maxY = math.max(minY, viewport.Y - halfH - 8)

    local current = Vector2.new(
        viewport.X * panel.Position.X.Scale + panel.Position.X.Offset,
        viewport.Y * panel.Position.Y.Scale + panel.Position.Y.Offset
    )

    local clamped = Vector2.new(
        math.clamp(current.X, minX, maxX),
        math.clamp(current.Y, minY, maxY)
    )

    panel.Position = UDim2.fromOffset(clamped.X, clamped.Y)
end

addConnection(workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
    task.defer(clampPanelToViewport)
end))

addConnection(RunService.RenderStepped:Connect(function()
    if uiVisible and not uiMinimized then
        clampPanelToViewport()
    end
end))

local function getCamera()
    return workspace.CurrentCamera
end

local function getCharacterHumanoid(character)
    if not character then
        return nil
    end
    return character:FindFirstChildOfClass("Humanoid")
end

-- A character can remain in Workspace for a short time after dying while the
-- game moves it to a death/respawn staging area.  We must treat that state as
-- invalid for camera locking, even if the Model and body parts still exist.
local function isHumanoidAlive(humanoid)
    if not humanoid then
        return false
    end

    if humanoid.Health <= 0 then
        return false
    end

    local state = humanoid:GetState()
    if state == Enum.HumanoidStateType.Dead then
        return false
    end

    return true
end

local function resetLockedTargetMotion()
    lockedTargetMotion.character = nil
    lockedTargetMotion.lastPosition = nil
    lockedTargetMotion.lastTime = 0
end

local function updateLockedTargetMotion(character)
    if not character then
        resetLockedTargetMotion()
        return false
    end

    local root = character:FindFirstChild("HumanoidRootPart")
    if not root or not root:IsA("BasePart") then
        root = character:FindFirstChild("Head")
    end
    if not root or not root:IsA("BasePart") then
        return false
    end

    local now = os.clock()
    local position = root.Position

    if lockedTargetMotion.character ~= character then
        lockedTargetMotion.character = character
        lockedTargetMotion.lastPosition = position
        lockedTargetMotion.lastTime = now
        return false
    end

    local previousPosition = lockedTargetMotion.lastPosition
    local previousTime = lockedTargetMotion.lastTime
    lockedTargetMotion.lastPosition = position
    lockedTargetMotion.lastTime = now

    if previousPosition and previousTime > 0 then
        local dt = now - previousTime
        local deltaY = position.Y - previousPosition.Y

        -- Hypershot-like respawn flow can move a dead character sharply below
        -- the map before the new character is spawned. Treat a large, sudden
        -- downward teleport as a staging/death transition.
        if dt > 0 and deltaY <= -30 and (deltaY / dt) <= -120 then
            return true
        end
    end

    -- Hard safety net for a corpse/staging model placed far below the normal
    -- playable area. This does not affect ordinary targets at normal map height.
    if position.Y < -150 then
        return true
    end

    return false
end

-- Non-mutating safety check used during target acquisition as well.
local function isLikelyDeathStagingCharacter(character)
    if not character then
        return true
    end

    local humanoid = getCharacterHumanoid(character)
    if not isHumanoidAlive(humanoid) then
        return true
    end

    local root = character:FindFirstChild("HumanoidRootPart")
        or character:FindFirstChild("Head")
    if not root or not root:IsA("BasePart") then
        return false
    end

    return root.Position.Y < -150
end

local function getAimPart(character)
    if not character then
        return nil
    end

    if targetPartMode == "Head" then
        local head = character:FindFirstChild("Head")
        if head and head:IsA("BasePart") then
            return head
        end

        -- Fallback for rigs without a Head part.
        local upperTorso = character:FindFirstChild("UpperTorso")
        if upperTorso and upperTorso:IsA("BasePart") then
            return upperTorso
        end

        local torso = character:FindFirstChild("Torso")
        if torso and torso:IsA("BasePart") then
            return torso
        end

        local root = character:FindFirstChild("HumanoidRootPart")
        if root and root:IsA("BasePart") then
            return root
        end
    else
        -- Body mode: prefer the torso, then the root.
        local upperTorso = character:FindFirstChild("UpperTorso")
        if upperTorso and upperTorso:IsA("BasePart") then
            return upperTorso
        end

        local torso = character:FindFirstChild("Torso")
        if torso and torso:IsA("BasePart") then
            return torso
        end

        local root = character:FindFirstChild("HumanoidRootPart")
        if root and root:IsA("BasePart") then
            return root
        end

        -- Fallback for custom rigs.
        for _, child in ipairs(character:GetChildren()) do
            if child:IsA("BasePart") and child.Name ~= "Head" then
                return child
            end
        end
    end

    -- Last-resort fallback.
    for _, child in ipairs(character:GetChildren()) do
        if child:IsA("BasePart") then
            return child
        end
    end

    return nil
end

local function sameTeam(player)
    if player == localPlayer then
        return true
    end

    -- If either side does not expose a Team, treat the target as an enemy.
    -- This is important for FFA games where all players commonly have no Team.
    if localPlayer.Team == nil or player.Team == nil then
        return false
    end

    return localPlayer.Team == player.Team
end

local function shouldTargetPlayer(player)
    if player == localPlayer then
        return false
    end

    if not player.Character then
        return false
    end

    -- When Team Check is enabled, teammates are excluded from target selection.
    -- In FFA games where Team is nil, sameTeam() returns false, so everyone remains valid.
    if teamCheckEnabled and sameTeam(player) then
        return false
    end

    return true
end

    status.TextColor3 = color or COLORS.subtext
end

local function createVisualLine(parent, color, thickness)
    local line = Instance.new("Frame")
    line.BackgroundColor3 = color or COLORS.white
    line.BackgroundTransparency = 0.05
    line.BorderSizePixel = 0
    line.AnchorPoint = Vector2.new(0.5, 0.5)
    line.Size = UDim2.fromOffset(thickness or 2, thickness or 2)
    line.Visible = false
    line.ZIndex = 2
    line.Parent = parent
    return line
end

local function setScreenLine(line, pointA, pointB, thickness)
    if not line then
        return
    end

    local delta = pointB - pointA
    local length = delta.Magnitude

    if length < 0.5 then
        line.Visible = false
        return
    end

    line.Position = UDim2.fromOffset((pointA.X + pointB.X) * 0.5, (pointA.Y + pointB.Y) * 0.5)
    line.Size = UDim2.fromOffset(length, thickness or 2)
    line.Rotation = math.deg(math.atan2(delta.Y, delta.X))
    line.Visible = true
end

local function getPartAny(character, names)
    for _, name in ipairs(names) do
        local part = character:FindFirstChild(name)
        if part and part:IsA("BasePart") then
            return part
        end
    end
    return nil
end

local function getSkeletonPairs(character)
    local head = getPartAny(character, {"Head"})
    local chest = getPartAny(character, {"UpperTorso", "Torso", "HumanoidRootPart"})
    local pelvis = getPartAny(character, {"LowerTorso", "HumanoidRootPart", "Torso"})

    local leftShoulder = getPartAny(character, {"LeftUpperArm", "Left Arm"})
    local leftElbow = getPartAny(character, {"LeftLowerArm", "Left Arm"})
    local leftHand = getPartAny(character, {"LeftHand", "Left Arm"})
    local rightShoulder = getPartAny(character, {"RightUpperArm", "Right Arm"})
    local rightElbow = getPartAny(character, {"RightLowerArm", "Right Arm"})
    local rightHand = getPartAny(character, {"RightHand", "Right Arm"})

    local leftHip = getPartAny(character, {"LeftUpperLeg", "Left Leg"})
    local leftKnee = getPartAny(character, {"LeftLowerLeg", "Left Leg"})
    local leftFoot = getPartAny(character, {"LeftFoot", "Left Leg"})
    local rightHip = getPartAny(character, {"RightUpperLeg", "Right Leg"})
    local rightKnee = getPartAny(character, {"RightLowerLeg", "Right Leg"})
    local rightFoot = getPartAny(character, {"RightFoot", "Right Leg"})

    return {
        {head, chest},
        {chest, pelvis},
        {chest, leftShoulder},
        {leftShoulder, leftElbow},
        {leftElbow, leftHand},
        {chest, rightShoulder},
        {rightShoulder, rightElbow},
        {rightElbow, rightHand},
        {pelvis, leftHip},
        {leftHip, leftKnee},
        {leftKnee, leftFoot},
        {pelvis, rightHip},
        {rightHip, rightKnee},
        {rightKnee, rightFoot},
    }
end

local function getEspAnchorPart(character)
    if not character then
        return nil
    end

    local preferred = {
        "Head",
        "UpperTorso",
        "Torso",
        "HumanoidRootPart",
    }

    for _, name in ipairs(preferred) do
        local part = character:FindFirstChild(name)
        if part and part:IsA("BasePart") then
            return part
        end
    end

    for _, child in ipairs(character:GetChildren()) do
        if child:IsA("BasePart") then
            return child
        end
    end

    return nil
end

local function getEspContainer(player)
    local playerGui = localPlayer:FindFirstChildOfClass("PlayerGui")
    if not playerGui then
        return nil
    end

    local name = espGuiName .. "_" .. player.UserId
    local folder = playerGui:FindFirstChild(name)
    if folder and folder:IsA("Folder") then
        return folder
    end

    folder = Instance.new("Folder")
    folder.Name = name
    folder.Parent = playerGui
    return folder
end

local function destroyPlayerEsp(player)
    local data = espObjects[player]
    if data then
        -- Only destroy ESP-owned instances. Never destroy the Character or
        -- its body parts stored as references in the state table.
        for _, key in ipairs({"highlight", "billboard", "folder"}) do
            local object = data[key]
            if typeof(object) == "Instance" then
                pcall(function()
                    object:Destroy()
                end)
            end
        end

        for _, key in ipairs({"boxLines", "skeletonLines"}) do
            local lines = data[key]
            if type(lines) == "table" then
                for _, line in ipairs(lines) do
                    if typeof(line) == "Instance" then
                        pcall(function()
                            line:Destroy()
                        end)
                    end
                end
            end
        end
    end

    espObjects[player] = nil

    -- Clean any stale objects left on the current character.
    local character = player and player.Character
    if character then
        local highlight = character:FindFirstChild(highlightName)
        if highlight then
            pcall(function() highlight:Destroy() end)
        end
    end

    local playerGui = localPlayer:FindFirstChildOfClass("PlayerGui")
    if playerGui and player then
        local folder = playerGui:FindFirstChild(espGuiName .. "_" .. player.UserId)
        if folder then
            pcall(function() folder:Destroy() end)
        end
    end
end

local function applyEspColors(player, highlight, tag, boxLines, skeletonLines)
    local teammate = sameTeam(player)
    local color = teammate and COLORS.blue or COLORS.red

    if highlight then
        if teammate then
            highlight.FillColor = COLORS.blue
            highlight.OutlineColor = COLORS.white
        else
            highlight.FillColor = COLORS.red
            highlight.OutlineColor = Color3.fromRGB(255, 230, 90)
        end
    end

    if tag then
        tag.TextColor3 = teammate and COLORS.blue or COLORS.white
    end

    if boxLines then
        for _, line in ipairs(boxLines) do
            line.BackgroundColor3 = color
        end
    end

    if skeletonLines then
        for _, line in ipairs(skeletonLines) do
            line.BackgroundColor3 = color
        end
    end
end

local function buildPlayerEsp(player, character)
    if player == localPlayer or not player.Parent then
        return false
    end

    if player.Character ~= character or not character or not character.Parent then
        return false
    end

    if not espEnabled then
        destroyPlayerEsp(player)
        return false
    end

    -- Team Check also controls ESP: teammates should not be visible through walls.
    if teamCheckEnabled and sameTeam(player) then
        destroyPlayerEsp(player)
        return false
    end

    local humanoid = getCharacterHumanoid(character)
    local anchor = getEspAnchorPart(character)
    if not humanoid or not anchor then
        return false
    end

    destroyPlayerEsp(player)

    local highlight = nil
    if espAuraEnabled then
        highlight = Instance.new("Highlight")
        highlight.Name = highlightName
        highlight.Adornee = character
        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        highlight.FillTransparency = 0.72
        highlight.OutlineTransparency = 0
        highlight.Enabled = true
        highlight.Parent = character
    end

    local folder = getEspContainer(player)
    if not folder then
        if highlight then highlight:Destroy() end
        return false
    end

    local billboard = nil
    local tag = nil
    if espNameEnabled then
        billboard = Instance.new("BillboardGui")
        billboard.Name = "ESPTag"
        billboard.Adornee = anchor
        billboard.Size = UDim2.fromOffset(170, 32)
        billboard.StudsOffset = Vector3.new(0, 2.7, 0)
        billboard.AlwaysOnTop = true
        billboard.MaxDistance = 2500
        billboard.ResetOnSpawn = false
        billboard.Parent = folder

        tag = Instance.new("TextLabel")
        tag.Name = "Name"
        tag.Size = UDim2.fromScale(1, 1)
        tag.BackgroundTransparency = 1
        tag.Text = player.DisplayName
        tag.Font = Enum.Font.GothamBold
        tag.TextSize = 11
        tag.TextStrokeTransparency = 0.25
        tag.Parent = billboard
    end

    local boxLines = {}
    if espBoxEnabled then
        for _ = 1, 4 do
            table.insert(boxLines, createVisualLine(espOverlay, COLORS.white, 2))
        end
    end

    local skeletonLines = {}
    if espSkeletonEnabled then
        for _ = 1, 14 do
            table.insert(skeletonLines, createVisualLine(espOverlay, COLORS.white, 2))
        end
    end

    applyEspColors(player, highlight, tag, boxLines, skeletonLines)

    espObjects[player] = {
        character = character,
        highlight = highlight,
        billboard = billboard,
        folder = folder,
        anchor = anchor,
        boxLines = boxLines,
        skeletonLines = skeletonLines,
    }

    return true
end

local function schedulePlayerEsp(player, character, reason)
    if player == localPlayer or not player.Parent then
        return
    end

    local previousToken = espBuildTokens[player] or 0
    local token = previousToken + 1
    espBuildTokens[player] = token

    task.spawn(function()
        local deadline = os.clock() + 8

        while os.clock() < deadline do
            if not espEnabled then
                return
            end

            if not player.Parent or player.Character ~= character or not character.Parent then
                return
            end

            if espBuildTokens[player] ~= token then
                return
            end

            local humanoid = getCharacterHumanoid(character)
            local anchor = getEspAnchorPart(character)

            if humanoid and anchor then
                if buildPlayerEsp(player, character) then
                    return
                end
            end

            task.wait(0.08)
        end
    end)
end

local function refreshPlayerEsp(player)
    if player == localPlayer then
        return
    end

    local character = player.Character
    destroyPlayerEsp(player)

    if character and character.Parent and espEnabled then
        schedulePlayerEsp(player, character, "refresh")
    end
end

local function refreshAllEsp()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer then
            refreshPlayerEsp(player)
        end
    end
end

local function auditPlayerEsp(player)
    if player == localPlayer or not espEnabled then
        return
    end

    local character = player.Character
    if not character or not character.Parent then
        return
    end

    if teamCheckEnabled and sameTeam(player) then
        destroyPlayerEsp(player)
        return
    end

    local data = espObjects[player]

    if not data or data.character ~= character then
        schedulePlayerEsp(player, character, "audit-missing")
        return
    end

    local needsRebuild = false

    if espAuraEnabled then
        local valid = data.highlight and data.highlight.Parent and data.highlight.Adornee == character
        if not valid then needsRebuild = true end
    elseif data.highlight then
        needsRebuild = true
    end

    if espNameEnabled then
        local valid = data.billboard and data.billboard.Parent and data.billboard.Adornee and data.billboard.Adornee.Parent
        if not valid then needsRebuild = true end
    elseif data.billboard then
        needsRebuild = true
    end

    if espBoxEnabled then
        if #((data.boxLines) or {}) ~= 4 then needsRebuild = true end
    elseif data.boxLines and #data.boxLines > 0 then
        needsRebuild = true
    end

    if espSkeletonEnabled then
        if #((data.skeletonLines) or {}) ~= 14 then needsRebuild = true end
    elseif data.skeletonLines and #data.skeletonLines > 0 then
        needsRebuild = true
    end

    if needsRebuild then
        schedulePlayerEsp(player, character, "audit-invalid")
        return
    end

    local tag = data.billboard and data.billboard:FindFirstChild("Name")
    applyEspColors(player, data.highlight, tag, data.boxLines, data.skeletonLines)
end

local function projectPart(camera, part)
    if not camera or not part or not part.Parent then
        return nil, false
    end

    local point, onScreen = camera:WorldToViewportPoint(part.Position)
    if not onScreen or point.Z <= 0 then
        return nil, false
    end

    return Vector2.new(point.X, point.Y), true
end

local function updatePlayerBox(data, camera)
    local lines = data.boxLines
    if not lines or #lines < 4 then
        return
    end

    local character = data.character
    if not character or not character.Parent then
        for _, line in ipairs(lines) do line.Visible = false end
        return
    end

    local cf, size = character:GetBoundingBox()
    local half = size * 0.5
    local corners = {
        cf:PointToWorldSpace(Vector3.new(-half.X, -half.Y, -half.Z)),
        cf:PointToWorldSpace(Vector3.new(-half.X,  half.Y, -half.Z)),
        cf:PointToWorldSpace(Vector3.new( half.X, -half.Y, -half.Z)),
        cf:PointToWorldSpace(Vector3.new( half.X,  half.Y, -half.Z)),
        cf:PointToWorldSpace(Vector3.new(-half.X, -half.Y,  half.Z)),
        cf:PointToWorldSpace(Vector3.new(-half.X,  half.Y,  half.Z)),
        cf:PointToWorldSpace(Vector3.new( half.X, -half.Y,  half.Z)),
        cf:PointToWorldSpace(Vector3.new( half.X,  half.Y,  half.Z)),
    }

    local minX, minY = math.huge, math.huge
    local maxX, maxY = -math.huge, -math.huge
    local visibleCount = 0

    for _, worldPoint in ipairs(corners) do
        local screenPoint, visible = projectPart(camera, {Position = worldPoint, Parent = character})
        if visible and screenPoint then
            visibleCount += 1
            minX = math.min(minX, screenPoint.X)
            minY = math.min(minY, screenPoint.Y)
            maxX = math.max(maxX, screenPoint.X)
            maxY = math.max(maxY, screenPoint.Y)
        end
    end

    if visibleCount == 0 or maxX <= minX or maxY <= minY then
        for _, line in ipairs(lines) do line.Visible = false end
        return
    end

    local topLeft = Vector2.new(minX, minY)
    local topRight = Vector2.new(maxX, minY)
    local bottomRight = Vector2.new(maxX, maxY)
    local bottomLeft = Vector2.new(minX, maxY)

    setScreenLine(lines[1], topLeft, topRight, 2)
    setScreenLine(lines[2], topRight, bottomRight, 2)
    setScreenLine(lines[3], bottomRight, bottomLeft, 2)
    setScreenLine(lines[4], bottomLeft, topLeft, 2)
end

local function updatePlayerSkeleton(data, camera)
    local lines = data.skeletonLines
    if not lines or #lines < 14 then
        return
    end

    local pairs = getSkeletonPairs(data.character)
    for index, pair in ipairs(pairs) do
        local line = lines[index]
        local a, b = pair[1], pair[2]
        if line and a and b then
            local pa, va = projectPart(camera, a)
            local pb, vb = projectPart(camera, b)
            if va and vb and pa and pb then
                setScreenLine(line, pa, pb, 2)
            else
                line.Visible = false
            end
        elseif line then
            line.Visible = false
        end
    end
end

local function updateAllVisuals()
    if not espEnabled then
        return
    end

    local camera = getCamera()
    if not camera then
        return
    end

    for player, data in pairs(espObjects) do
        if player.Parent and data and data.character == player.Character then
            if espBoxEnabled then
                updatePlayerBox(data, camera)
            elseif data.boxLines then
                for _, line in ipairs(data.boxLines) do line.Visible = false end
            end

            if espSkeletonEnabled then
                updatePlayerSkeleton(data, camera)
            elseif data.skeletonLines then
                for _, line in ipairs(data.skeletonLines) do line.Visible = false end
            end
        end
    end
end

-- =========================================================
-- 8. Target selection
-- =========================================================

-- Select exactly ONE target.
-- 1) Only candidates inside the configured FOV are considered.
-- 2) Exactly one candidate is returned: the aim point closest to screen center.
-- 3) If two are effectively tied on screen, use world distance as a tie-breaker.
-- Once acquired, the lock stays on this single target until release/invalid/death.
local function selectSingleTargetInFov()
    local camera = getCamera()
    if not camera then
        return nil, nil
    end

    local closestCharacter = nil
    local closestPart = nil
    local shortestScreenDistance = math.huge
    local closestWorldDistance = math.huge
    local viewportCenter = Vector2.new(
        camera.ViewportSize.X * 0.5,
        camera.ViewportSize.Y * 0.5
    )

    for _, player in ipairs(Players:GetPlayers()) do
        if shouldTargetPlayer(player) then
            local character = player.Character
            local humanoid = getCharacterHumanoid(character)
            local aimPart = getAimPart(character)

            if character and humanoid and isHumanoidAlive(humanoid) and aimPart
                and aimPart:IsDescendantOf(workspace)
                and not isLikelyDeathStagingCharacter(character) then
                local screenPoint, onScreen = camera:WorldToViewportPoint(aimPart.Position)

                if onScreen and screenPoint.Z > 0 then
                    local screenPos = Vector2.new(screenPoint.X, screenPoint.Y)
                    local screenDistance = (screenPos - viewportCenter).Magnitude

                    if screenDistance <= currentRadius then
                        -- Primary rule: closest to the reticle center.
                        -- Secondary rule: if two aim points land almost exactly
                        -- together on-screen, prefer the physically closer target.
                        local worldDistance = (aimPart.Position - camera.CFrame.Position).Magnitude
                        local isBetter = screenDistance < shortestScreenDistance
                            or (math.abs(screenDistance - shortestScreenDistance) < 0.01
                                and worldDistance < closestWorldDistance)

                        if isBetter then
                            shortestScreenDistance = screenDistance
                            closestWorldDistance = worldDistance
                            closestCharacter = character
                            closestPart = aimPart
                        end
                    end
                end
            end
        end
    end

    return closestCharacter, closestPart
end

local function isTargetInsideFov(character)
    local camera = getCamera()
    if not camera or not character then
        return false
    end

    local aimPart = getAimPart(character)
    if not aimPart or not aimPart:IsDescendantOf(workspace) then
        return false
    end

    local screenPoint, onScreen = camera:WorldToViewportPoint(aimPart.Position)
    if not onScreen or screenPoint.Z <= 0 then
        return false
    end

    local viewportCenter = Vector2.new(camera.ViewportSize.X * 0.5, camera.ViewportSize.Y * 0.5)
    local screenPos = Vector2.new(screenPoint.X, screenPoint.Y)
    return (screenPos - viewportCenter).Magnitude <= currentRadius
end

local function isValidLockedTarget(character)
    if not character or not character.Parent then
        return false
    end

    local player = Players:GetPlayerFromCharacter(character)
    if not player or player == localPlayer then
        return false
    end

    if not shouldTargetPlayer(player) then
        return false
    end

    local humanoid = getCharacterHumanoid(character)
    local aimPart = getAimPart(character)

    if not isHumanoidAlive(humanoid) then
        return false
    end

    if not aimPart or not aimPart:IsDescendantOf(workspace) then
        return false
    end

    -- Extra safety for games that teleport a dead/staging character below the
    -- map before Roblox reports the final death state.
    if updateLockedTargetMotion(character) then
        return false
    end

    -- FOV is used to ACQUIRE a target. Once a target is locked, keep that
    -- target while the lock button remains held. Mouse movement therefore does
    -- not break the lock. The target is still rejected if it dies, disappears,
    -- becomes invalid, fails Team Check, or is moved into a death/staging area.
    return true
end

local function isSupportedMouseButton(inputType)
    return inputType == Enum.UserInputType.MouseButton1
        or inputType == Enum.UserInputType.MouseButton2
        or inputType == Enum.UserInputType.MouseButton3
        or inputType == Enum.UserInputType.MouseButton4
end

local function getMouseButtonLabel(inputType)
    if inputType == Enum.UserInputType.MouseButton1 then
        return "MOUSE1"
    elseif inputType == Enum.UserInputType.MouseButton2 then
        return "MOUSE2"
    elseif inputType == Enum.UserInputType.MouseButton3 then
        return "MOUSE3"
    elseif inputType == Enum.UserInputType.MouseButton4 then
        return "MOUSE4"
    end

    return "MOUSE"
end

local function getLockBindLabel()
    if holdBindKind == "Mouse" then
        return getMouseButtonLabel(holdMouseButton)
    end

    return holdKey.Name
end

local function lockInputMatches(input)
    if holdBindKind == "Mouse" then
        return isSupportedMouseButton(input.UserInputType)
            and input.UserInputType == holdMouseButton
    end

    return input.UserInputType == Enum.UserInputType.Keyboard
        and input.KeyCode == holdKey
end

local function updateKeyUi()
    holdKeyButton.Text = "LOCK KEY  [" .. getLockBindLabel() .. "]"
    hideKeyButton.Text = "UI KEY  [" .. hideKey.Name .. "]"
end

local function updateFovUi(alpha)
    alpha = math.clamp(alpha, 0, 1)
    currentRadius = math.floor(minRadius + (maxRadius - minRadius) * alpha + 0.5)
    fovFill.Size = UDim2.new(alpha, 0, 1, 0)
    fovKnob.Position = UDim2.new(alpha, -9, 0, -4)
    fovText.Text = "FOV  " .. currentRadius
    fovValue.Text = tostring(currentRadius)
    fovCircle.Size = UDim2.fromOffset(currentRadius * 2, currentRadius * 2)
end

local function updateFovFromInput(input)
    local width = fovSlider.AbsoluteSize.X
    if width <= 0 then
        return
    end

    local x = fovSlider.AbsolutePosition.X
    local alpha = (input.Position.X - x) / width
    updateFovUi(alpha)
end

local function getSmoothPercent(level)
    if level <= 9 then
        return 100 - (level * 10)
    end

    return 5
end

local function getSmoothFactor()
    -- 0 = full snap, higher levels reduce the amount of rotation applied per frame.
    if smoothLevel <= 9 then
        return (10 - smoothLevel) / 10
    end

    return 0.05
end

local function updateSmoothUiFromLevel(level)
    smoothLevel = math.clamp(math.floor(level + 0.5), minSmoothLevel, maxSmoothLevel)
    local alpha = smoothLevel / maxSmoothLevel
    local percent = getSmoothPercent(smoothLevel)

    smoothFill.Size = UDim2.new(alpha, 0, 1, 0)
    smoothKnob.Position = UDim2.new(alpha, -9, 0, -4)
    smoothValue.Text = tostring(smoothLevel) .. " • " .. tostring(percent) .. "%"
end

local function updateSmoothFromInput(input)
    local width = smoothSlider.AbsoluteSize.X
    if width <= 0 then
        return
    end

    local x = smoothSlider.AbsolutePosition.X
    local alpha = math.clamp((input.Position.X - x) / width, 0, 1)
    local level = alpha * maxSmoothLevel
    updateSmoothUiFromLevel(level)
end

-- =========================================================
-- 10. Player monitoring (respawn / rig replacement safe)
-- =========================================================

local function monitorPlayer(player)
    if player == localPlayer then
        return
    end

    disconnectList(playerConnections[player])
    playerConnections[player] = {}
    local list = playerConnections[player]

    table.insert(list, player.CharacterAdded:Connect(function(character)
        -- Kill the old ESP immediately, then wait for the new rig to become usable.
        destroyPlayerEsp(player)
        if lockedTarget and Players:GetPlayerFromCharacter(lockedTarget) == player then
            lockedTarget = nil
        end
        schedulePlayerEsp(player, character, "CharacterAdded")

        local function watchHumanoid(humanoid)
            if not humanoid or not humanoid:IsA("Humanoid") then
                return
            end

            -- Hypershot-like games may keep a dead Character in Workspace while
            -- waiting for the real respawn. Drop the lock immediately on Died
            -- instead of letting the camera follow the staged/dead body.
            table.insert(list, humanoid.Died:Connect(function()
                if lockedTarget == character then
                    lockedTarget = nil
                    resetLockedTargetMotion()
                    if isHoldingKey then
                        setStatus("TARGET DEAD - SEARCHING", COLORS.orange)
                    end
                end
            end))

            table.insert(list, humanoid:GetPropertyChangedSignal("Health"):Connect(function()
                if humanoid.Health <= 0 and lockedTarget == character then
                    lockedTarget = nil
                    resetLockedTargetMotion()
                    if isHoldingKey then
                        setStatus("TARGET DEAD - SEARCHING", COLORS.orange)
                    end
                end
            end))

            table.insert(list, humanoid.StateChanged:Connect(function(_, newState)
                if newState == Enum.HumanoidStateType.Dead and lockedTarget == character then
                    lockedTarget = nil
                    resetLockedTargetMotion()
                    resetLockedTargetMotion()
                    if isHoldingKey then
                        setStatus("TARGET DEAD - SEARCHING", COLORS.orange)
                    end
                end
            end))
        end

        local humanoid = getCharacterHumanoid(character)
        if humanoid then
            watchHumanoid(humanoid)
        end

        -- Some games finish inserting body parts shortly after CharacterAdded.
        table.insert(list, character.ChildAdded:Connect(function(child)
            if player.Character ~= character then
                return
            end

            if child:IsA("Humanoid") then
                watchHumanoid(child)
                schedulePlayerEsp(player, character, "HumanoidAdded")
            elseif child:IsA("BasePart") then
                schedulePlayerEsp(player, character, "ChildAdded")
            end
        end))
    end))

    if player.CharacterRemoving then
        table.insert(list, player.CharacterRemoving:Connect(function(character)
            if lockedTarget == character then
                lockedTarget = nil
                resetLockedTargetMotion()
            end

            local data = espObjects[player]
            if data and data.character == character then
                destroyPlayerEsp(player)
            end
        end))
    end

    -- CharacterAppearanceLoaded is an additional signal for games that build
    -- their visible rig in stages.
    if player.CharacterAppearanceLoaded then
        table.insert(list, player.CharacterAppearanceLoaded:Connect(function(character)
            if player.Character == character then
                schedulePlayerEsp(player, character, "CharacterAppearanceLoaded")
            end
        end))
    end

    table.insert(list, player:GetPropertyChangedSignal("Team"):Connect(function()
        -- Re-evaluate both target eligibility and ESP visibility immediately.
        refreshPlayerEsp(player)
    end))

    if player.Character then
        refreshPlayerEsp(player)
    end
end

    local lastAudit = sharedEnv.__AutoGameMenuEspAuditTime or 0
    if now - lastAudit < 0.20 then
        return
    end
    sharedEnv.__AutoGameMenuEspAuditTime = now

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer then
            auditPlayerEsp(player)
        end
    end

addConnection(UserInputService.InputBegan:Connect(function(input, gameProcessed)
    -- Lock binding can be any keyboard key OR Mouse1-4.
    if waitingForHoldKey then
        if input.UserInputType == Enum.UserInputType.Keyboard then
            holdBindKind = "Keyboard"
            holdKey = input.KeyCode
        elseif isSupportedMouseButton(input.UserInputType) then
            holdBindKind = "Mouse"
            holdMouseButton = input.UserInputType
        else
            return
        end

        waitingForHoldKey = false
        waitingForHideKey = false
        holdKeyButton.BackgroundColor3 = COLORS.panel2
        updateKeyUi()
        setStatus("LOCK KEY SET TO " .. getLockBindLabel(), COLORS.green)
        return
    end

    if waitingForHideKey then
        if input.UserInputType ~= Enum.UserInputType.Keyboard then
            return
        end

        hideKey = input.KeyCode
        waitingForHideKey = false
        waitingForHoldKey = false
        hideKeyButton.BackgroundColor3 = COLORS.panel2
        updateKeyUi()
        setStatus("UI KEY SET TO " .. hideKey.Name, COLORS.green)
        return
    end

    -- Hide/show UI still uses the keyboard-only UI key.
    if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == hideKey then
        setUiVisible(not uiVisible)
        return
    end

    if gameProcessed then
        return
    end

    if lockEnabled and lockInputMatches(input) and not isHoldingKey then
        isHoldingKey = true
        lockedTarget = nil

        local target = selectSingleTargetInFov()
        if target then
            lockedTarget = target
            resetLockedTargetMotion()
            updateLockedTargetMotion(lockedTarget)
            setStatus("TARGET LOCKED", COLORS.green)
        else
            isHoldingKey = false
            setStatus("NO TARGET IN FOV", COLORS.orange)
        end
    end
end))

    if lockInputMatches(input) then
        isHoldingKey = false
        lockedTarget = nil
        resetLockedTargetMotion()
        if lockEnabled then
            setStatus("LOCK READY", COLORS.subtext)
        end
    end
end))

-- 12. Camera lock
-- =========================================================

addConnection(RunService.RenderStepped:Connect(function()
    if not lockEnabled or not isHoldingKey then
        return
    end

    -- The FOV is checked when acquiring the target. After acquisition, keep
    -- the same target until the lock button is released or the target becomes
    -- invalid/dead. This prevents normal mouse movement from dropping lock.
    -- Death/respawn staging is checked independently so the camera never
    -- follows a corpse that the game temporarily moves below the map.
    if not lockedTarget or not isValidLockedTarget(lockedTarget) then
        local replacement = selectSingleTargetInFov()
        if replacement then
            lockedTarget = replacement
            resetLockedTargetMotion()
            updateLockedTargetMotion(lockedTarget)
            setStatus("TARGET IN FOV", COLORS.green)
        else
            -- Nothing is currently inside the configured FOV. Do not rotate
            -- the camera toward a dead/off-screen player.
            lockedTarget = nil
            resetLockedTargetMotion()
            return
        end
    end

    local camera = getCamera()
    if not camera then
        return
    end

    local targetHumanoid = getCharacterHumanoid(lockedTarget)
    if not isHumanoidAlive(targetHumanoid) or isLikelyDeathStagingCharacter(lockedTarget) then
        lockedTarget = nil
        resetLockedTargetMotion()
        return
    end

    local aimPart = getAimPart(lockedTarget)
    if not aimPart or not aimPart:IsDescendantOf(workspace) then
        lockedTarget = nil
        resetLockedTargetMotion()
        return
    end

    local cameraPosition = camera.CFrame.Position
    local targetPosition = aimPart.Position
    local desiredCFrame = CFrame.lookAt(cameraPosition, targetPosition)
    local smoothFactor = getSmoothFactor()

    -- Keep camera position fixed and smoothly rotate toward the target.
    if smoothFactor >= 0.999 then
        camera.CFrame = desiredCFrame
    else
        camera.CFrame = camera.CFrame:Lerp(desiredCFrame, smoothFactor)
    end
end))

-- 14. Cleanup
-- =========================================================

local cleanedUp = false

local function cleanup()
    if cleanedUp then
        return
    end

    cleanedUp = true
    lockEnabled = false
    isHoldingKey = false
    lockedTarget = nil
    isDraggingFov = false
    waitingForHoldKey = false
    waitingForHideKey = false

    disconnectAll()

    if screenGui then
        pcall(function()
            screenGui:Destroy()
        end)
    end

    if sharedEnv.__AutoGameMenuCleanup == cleanup then
        sharedEnv.__AutoGameMenuCleanup = nil
    end
end

sharedEnv.__AutoGameMenuCleanup = cleanup
addConnection(screenGui.Destroying:Connect(function()
    pcall(cleanup)
end))
